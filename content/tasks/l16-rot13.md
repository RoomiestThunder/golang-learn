title: Reader-декоратор: ROT13
level: medium

@@ statement
Напиши `NewRot13Reader(r io.Reader) io.Reader` — обёртку, которая читает из `r` и применяет шифр ROT13 к **латинским** буквам (`a↔n`, `A↔N`, …); остальные байты (цифры, знаки, кириллица) остаются как есть.

Требования:
- не загружай весь поток в память (преобразуй то, что вернул вложенный `Read`, на месте);
- корректно возвращай `n` и ошибку (включая `io.EOF`) от вложенного ридера;
- должна работать с «капризным» источником, отдающим по одному байту (`iotest.OneByteReader`).

Это тот самый паттерн «декоратор над интерфейсом», на котором построены `gzip.NewReader`, `bufio.NewReader` и т. д.
@@ starter
package solution

import "io"

func NewRot13Reader(r io.Reader) io.Reader {
	return r
}
@@ tests
package solution

import (
	"io"
	"strings"
	"testing"
	"testing/iotest"
)

func TestRot13(t *testing.T) {
	in := "Hello, Мир! abc XYZ 123 nopqrstuvwxyz"
	want := "Uryyb, Мир! nop KLM 123 abcdefghijklm"
	for name, src := range map[string]io.Reader{
		"обычный":  strings.NewReader(in),
		"по байту": iotest.OneByteReader(strings.NewReader(in)),
		"данные+EOF": iotest.DataErrReader(strings.NewReader(in)),
	} {
		got, err := io.ReadAll(NewRot13Reader(src))
		if err != nil || string(got) != want {
			t.Errorf("%s: got %q, %v; want %q", name, got, err, want)
		}
	}
}

func TestRot13Involution(t *testing.T) {
	in := strings.Repeat("The Quick Brown Fox. ", 5000)
	twice, _ := io.ReadAll(NewRot13Reader(NewRot13Reader(strings.NewReader(in))))
	if string(twice) != in {
		t.Error("двойное применение ROT13 должно вернуть исходный текст")
	}
}
@@ solution
package solution

import "io"

type rot13Reader struct{ r io.Reader }

func NewRot13Reader(r io.Reader) io.Reader { return rot13Reader{r} }

func (x rot13Reader) Read(p []byte) (int, error) {
	n, err := x.r.Read(p)
	for i := 0; i < n; i++ {
		c := p[i]
		switch {
		case c >= 'a' && c <= 'z':
			p[i] = 'a' + (c-'a'+13)%26
		case c >= 'A' && c <= 'Z':
			p[i] = 'A' + (c-'A'+13)%26
		}
	}
	return n, err
}
@@ hint
Структура с полем `r io.Reader` и методом `Read(p []byte) (int, error)`: вызови `r.Read(p)`, преобразуй `p[:n]`, верни `n, err`.
